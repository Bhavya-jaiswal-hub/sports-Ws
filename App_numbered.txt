34	function useWebSocket() {
35	  const wsRef = useRef(null);
36	  const [connected, setConnected] = useState(false);
37	  const [apiCount, setApiCount] = useState(0);
38	  const subscribedRef = useRef(new Set());
39	  const listenersRef = useRef([]);
40	
41	  const addListener = useCallback((fn) => {
42	    listenersRef.current.push(fn);
43	    return () => {
44	      listenersRef.current = listenersRef.current.filter((f) => f !== fn);
45	    };
46	  }, []);
47	
48	  const connect = useCallback(() => {
49	    if (wsRef.current?.readyState === WebSocket.OPEN) return;
50	    const ws = new WebSocket(WS_URL);
51	    wsRef.current = ws;
52	    ws.onopen = () => setConnected(true);
53	    ws.onclose = () => {
54	      setConnected(false);
55	      setTimeout(connect, 3000);
56	    };
57	    ws.onerror = () => ws.close();
58	    ws.onmessage = (e) => {
59	      try {
60	        const msg = JSON.parse(e.data);
61	        listenersRef.current.forEach((fn) => fn(msg));
62	      } catch {}
63	    };
64	  }, []);
65	
66	  useEffect(() => {
67	    connect();
68	    return () => wsRef.current?.close();
69	  }, [connect]);
70	
71	  const subscribe = useCallback((matchId) => {
72	    if (subscribedRef.current.has(matchId)) return;
73	    if (wsRef.current?.readyState === WebSocket.OPEN) {
74	      wsRef.current.send(JSON.stringify({ type: "subscribe", matchId }));
75	      subscribedRef.current.add(matchId);
76	    }
77	  }, []);
78	
79	  const unsubscribe = useCallback((matchId) => {
80	    if (!subscribedRef.current.has(matchId)) return;
81	    if (wsRef.current?.readyState === WebSocket.OPEN) {
82	      wsRef.current.send(JSON.stringify({ type: "unsubscribe", matchId }));
83	      subscribedRef.current.delete(matchId);
84	    }
85	  }, []);
86	
87	  const incrementApi = useCallback(() => setApiCount((c) => c + 1), []);
88	
89	  return { connected, apiCount, incrementApi, subscribe, unsubscribe, addListener };
90	}
